```typescript
// src/components/server-status/CopyIPButton.tsx
// Client Component — clipboard API is browser-only
'use client';
import { useState } from 'react';
import { toast } from 'sonner';
import { Button } from '@/components/ui/button';
import { Copy, Check } from 'lucide-react';

interface CopyIPButtonProps {
  ip: string;
}

export function CopyIPButton({ ip }: CopyIPButtonProps) {
  const [copied, setCopied] = useState(false);

  const handleCopy = async () => {
    await navigator.clipboard.writeText(ip);
    setCopied(true);
    toast.success('IP copiado!', { description: ip });
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <Button
      onClick={handleCopy}
      variant="outline"
      size="sm"
      data-testid="copy-ip-button"
      className="border-white/30 text-white hover:bg-white/10 gap-2 font-mono text-xs"
    >
      {copied ? <Check className="h-3 w-3 text-brand-online" /> : <Copy className="h-3 w-3" />}
      {copied ? 'Copiado!' : ip}
    </Button>
  );
}
```

**Step 1 — Create `src/components/banned-items/BannedItemsTable.tsx`:**

```typescript
// src/components/banned-items/BannedItemsTable.tsx
// Client Component — uses useState for search filter
'use client';
import { useState } from 'react';
import { Input } from '@/components/ui/input';
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from '@/components/ui/table';
import type { BannedItem } from '@/types';

interface BannedItemsTableProps {
  items: BannedItem[];
}

export function BannedItemsTable({ items }: BannedItemsTableProps) {
  const [query, setQuery] = useState('');

  const filtered = items.filter(
    (item) =>
      item.name.toLowerCase().includes(query.toLowerCase()) ||
      item.category.toLowerCase().includes(query.toLowerCase()) ||
      item.reason.toLowerCase().includes(query.toLowerCase())
  );

  return (
    <div className="space-y-4">
      <Input
        data-testid="banned-items-search"
        placeholder="Buscar item, categoria ou razão..."
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        className="bg-brand-surface border-white/20 text-white placeholder:text-white/40 max-w-sm"
      />

      {filtered.length === 0 ? (
        <div
          data-testid="banned-items-empty"
          className="text-center py-12 text-white/50"
        >
          Nenhum item encontrado para &quot;{query}&quot;.
        </div>
      ) : (
        <div className="rounded-card border border-white/10 overflow-hidden">
          <Table>
            <TableHeader>
              <TableRow className="border-white/10 hover:bg-transparent">
                <TableHead className="text-white/70">Item</TableHead>
                <TableHead className="text-white/70">Categoria</TableHead>
                <TableHead className="text-white/70">Razão do Ban</TableHead>
              </TableRow>
            </TableHeader>
            <TableBody>
              {filtered.map((item) => (
                <TableRow
                  key={item.name}
                  data-testid={`banned-item-row-${item.name.replace(/\s+/g, '-').toLowerCase()}`}
                  className="border-white/10 hover:bg-brand-surface"
                >
                  <TableCell className="font-medium text-white">{item.name}</TableCell>
                  <TableCell>
                    <span className="text-xs bg-brand-card text-brand-blue px-2 py-1 rounded-full">
                      {item.category}
                    </span>
                  </TableCell>
                  <TableCell className="text-white/70 text-sm">{item.reason}</TableCell>
                </TableRow>
              ))}
            </TableBody>
          </Table>
        </div>
      )}

      {/* Item count */}
      <p className="text-xs text-white/40">
        {filtered.length} {filtered.length === 1 ? 'item' : 'itens'} encontrado{filtered.length !== 1 ? 's' : ''}
        {query && ` para "${query}"`}
      </p>
    </div>
  );
}
```

// tests/banned-items.test.tsx
import { render, screen, fireEvent } from '@testing-library/react';
import { describe, it, expect } from 'vitest';
import { BannedItemsTable } from '@/components/banned-items/BannedItemsTable';

const sampleItems = [
  { name: 'Bedrock', reason: 'Item de admin', category: 'Bloco' },
  { name: 'Command Block', reason: 'Executa comandos', category: 'Mecanismo' },
  { name: 'TNT Minecart', reason: 'Dano massivo', category: 'Veículo' },
];

describe('BannedItemsTable', () => {
  it('renders all items when search is empty', () => {
    render(<BannedItemsTable items={sampleItems} />);
    expect(screen.getByText('Bedrock')).toBeInTheDocument();
    expect(screen.getByText('Command Block')).toBeInTheDocument();
    expect(screen.getByText('TNT Minecart')).toBeInTheDocument();
  });

  it('filters items by name on search input', () => {
    render(<BannedItemsTable items={sampleItems} />);
    const search = screen.getByTestId('banned-items-search');
    fireEvent.change(search, { target: { value: 'Bedrock' } });
    expect(screen.getByText('Bedrock')).toBeInTheDocument();
    expect(screen.queryByText('Command Block')).not.toBeInTheDocument();
    expect(screen.queryByText('TNT Minecart')).not.toBeInTheDocument();
  });

  it('filters items by category', () => {
    render(<BannedItemsTable items={sampleItems} />);
    const search = screen.getByTestId('banned-items-search');
    fireEvent.change(search, { target: { value: 'Mecanismo' } });
    expect(screen.getByText('Command Block')).toBeInTheDocument();
    expect(screen.queryByText('Bedrock')).not.toBeInTheDocument();
  });

  it('shows empty state when no items match the query', () => {
    render(<BannedItemsTable items={sampleItems} />);
    const search = screen.getByTestId('banned-items-search');
    fireEvent.change(search, { target: { value: 'xyz-item-que-nao-existe' } });
    expect(screen.getByTestId('banned-items-empty')).toBeInTheDocument();
  });

  it('search is case-insensitive', () => {
    render(<BannedItemsTable items={sampleItems} />);
    const search = screen.getByTestId('banned-items-search');
    fireEvent.change(search, { target: { value: 'bedrock' } });
    expect(screen.getByText('Bedrock')).toBeInTheDocument();
  });
});
