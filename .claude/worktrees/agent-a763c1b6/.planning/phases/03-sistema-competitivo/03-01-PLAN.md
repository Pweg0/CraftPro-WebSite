```typescript
import { db } from "@/lib/db"
import { arenaPlayers } from "@/lib/db/schema"
import { desc, eq, sql } from "drizzle-orm"

export async function getLeaderboard(page = 1, pageSize = 20) {
  const ranked = await db
    .select({
      uuid: arenaPlayers.uuid,
      name: arenaPlayers.name,
      elo: arenaPlayers.elo,
      wins: arenaPlayers.wins,
      losses: arenaPlayers.losses,
      division: arenaPlayers.division,
      placementComplete: arenaPlayers.placementComplete,
    })
    .from(arenaPlayers)
    .where(eq(arenaPlayers.placementComplete, 1))
    .orderBy(desc(arenaPlayers.elo))

  const unranked = await db
    .select({
      uuid: arenaPlayers.uuid,
      name: arenaPlayers.name,
      elo: arenaPlayers.elo,
      wins: arenaPlayers.wins,
      losses: arenaPlayers.losses,
      division: arenaPlayers.division,
      placementComplete: arenaPlayers.placementComplete,
    })
    .from(arenaPlayers)
    .where(eq(arenaPlayers.placementComplete, 0))
    .orderBy(desc(arenaPlayers.elo))

  return { ranked, unranked }
}

export async function getTop10Wins() {
  return db
    .select({
      uuid: arenaPlayers.uuid,
      name: arenaPlayers.name,
      elo: arenaPlayers.elo,
      wins: arenaPlayers.wins,
      losses: arenaPlayers.losses,
      division: arenaPlayers.division,
      placementComplete: arenaPlayers.placementComplete,
    })
    .from(arenaPlayers)
    .where(eq(arenaPlayers.placementComplete, 1))
    .orderBy(desc(arenaPlayers.wins))
    .limit(10)
}
```

```typescript
import { db } from "@/lib/db"
import { arenaPlayers, arenaMatches } from "@/lib/db/schema"
import { eq, or, desc, count } from "drizzle-orm"

export async function getPlayerProfile(uuid: string) {
  const [player] = await db
    .select()
    .from(arenaPlayers)
    .where(eq(arenaPlayers.uuid, uuid))
    .limit(1)
  return player ?? null
}

export async function getPlayerMatches(uuid: string, limit = 20, offset = 0) {
  return db
    .select({
      matchId: arenaMatches.matchId,
      arenaName: arenaMatches.arenaName,
      reason: arenaMatches.reason,
      startedAt: arenaMatches.startedAt,
      durationMs: arenaMatches.durationMs,
      winnerUuid: arenaMatches.winnerUuid,
      winnerName: arenaMatches.winnerName,
      loserUuid: arenaMatches.loserUuid,
      loserName: arenaMatches.loserName,
      winnerEloBefore: arenaMatches.winnerEloBefore,
      winnerEloAfter: arenaMatches.winnerEloAfter,
      winnerDelta: arenaMatches.winnerDelta,
      loserEloBefore: arenaMatches.loserEloBefore,
      loserEloAfter: arenaMatches.loserEloAfter,
      loserDelta: arenaMatches.loserDelta,
      winnerDivision: arenaMatches.winnerDivision,
      loserDivision: arenaMatches.loserDivision,
    })
    .from(arenaMatches)
    .where(or(
      eq(arenaMatches.winnerUuid, uuid),
      eq(arenaMatches.loserUuid, uuid)
    ))
    .orderBy(desc(arenaMatches.startedAt))
    .limit(limit)
    .offset(offset)
}

export async function getPlayerMatchCount(uuid: string) {
  const [result] = await db
    .select({ total: count() })
    .from(arenaMatches)
    .where(or(
      eq(arenaMatches.winnerUuid, uuid),
      eq(arenaMatches.loserUuid, uuid)
    ))
  return result?.total ?? 0
}
```

```typescript
export function calcWinRate(wins: number, losses: number): number {
  const total = wins + losses
  if (total === 0) return 0
  return Math.round((wins / total) * 100)
```

**1. Create `src/components/ranking/TierBadge.tsx`:**

```typescript
export const DIVISION_CONFIG = {
  UNRANKED:  { label: "Em Classificacao", color: "#888888" },
  BRONZE:    { label: "Bronze",           color: "#cd7f32" },
  SILVER:    { label: "Prata",            color: "#c0c0c0" },
  GOLD:      { label: "Ouro",             color: "#f59e0b" },
  PLATINUM:  { label: "Platina",          color: "#00d4ff" },
  DIAMOND:   { label: "Diamante",         color: "#b9f2ff" },
  MASTER:    { label: "Master",           color: "#ff4444" },
} as const

export type Division = keyof typeof DIVISION_CONFIG

interface TierBadgeProps {
  division: string
  size?: "sm" | "md"
}

export function TierBadge({ division, size = "sm" }: TierBadgeProps) {
  const key = (division?.toUpperCase() ?? "UNRANKED") as Division
  const config = DIVISION_CONFIG[key] ?? DIVISION_CONFIG.UNRANKED
  const sizeClasses = size === "sm"
    ? "px-2 py-0.5 text-xs"
    : "px-3 py-1 text-sm"

  return (
    <span
      className={`inline-flex items-center rounded-full font-semibold border whitespace-nowrap ${sizeClasses}`}
      style={{
        color: config.color,
        borderColor: `${config.color}60`,
        backgroundColor: `${config.color}15`,
      }}
    >
      {config.label}
    </span>
  )
}
```

This is a Server-safe component (no 'use client' needed). It uses inline styles for dynamic colors because the 7 division colors are defined by the mod spec and cannot be tokenized as Tailwind classes.

The `division` prop accepts `string` (not strict `Division`) because database values come as plain strings. The component falls back to UNRANKED config for any unknown value.
