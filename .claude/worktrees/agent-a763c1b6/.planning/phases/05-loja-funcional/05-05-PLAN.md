```typescript
/**
 * Resolves the RCON command to execute for a shop item purchase.
 * Per D-03: admin does NOT write commands manually — system generates from type.
 */
export function resolveRconCommand(
  item: { type: string | null; rconTemplate: string | null; name: string; duration: number | null },
  minecraftName: string
): string {
  // If admin provided a custom template, use it (future-proofing)
  if (item.rconTemplate) {
    return item.rconTemplate
      .replace(/\{player\}/g, minecraftName)
      .replace(/\{duration\}/g, String(item.duration ?? 30))
  }

  switch (item.type) {
    case 'rank': {
      const group = item.name.toLowerCase().replace(/\s+/g, '-')
      if (item.duration) {
        // LuckPerms temporary group: lp user <player> parent addtemp <group> <duration>d
        return `lp user ${minecraftName} parent addtemp ${group} ${item.duration}d`
      }
      return `lp user ${minecraftName} parent add ${group}`
    }
    case 'key':
      // Give a tripwire hook (key item) — server-side plugin maps to crate key
      return `give ${minecraftName} minecraft:tripwire_hook 1`
    case 'cosmetic': {
      const perm = item.name.toLowerCase().replace(/\s+/g, '-')
      return `lp user ${minecraftName} permission set craftpro.cosmetic.${perm} true`
    }
    default:
      throw new Error(`Unknown item type: ${item.type} for item "${item.name}"`)
  }
}
```

```typescript
import crypto from 'crypto'
import { NextResponse } from 'next/server'
import { db } from '@/lib/db'
import { orders, shopItems, users, deliveryQueue } from '@/lib/db/schema'
import { eq } from 'drizzle-orm'
import { resolveRconCommand } from '@/lib/shop/rcon'

export const runtime = 'nodejs'

/**
 * D-13: Validate MercadoPago HMAC-SHA256 signature.
 * x-signature format: "ts=1234567890,v1=abc123..."
 * Manifest: "id:{data.id};request-id:{x-request-id};ts:{ts};"
 */
function verifySignature(dataId: string, xSignature: string, xRequestId: string): boolean {
  const secret = process.env.MP_WEBHOOK_SECRET
  if (!secret) {
    console.error('[webhook] MP_WEBHOOK_SECRET not configured')
    return false
  }

  const parts: Record<string, string> = {}
  xSignature.split(',').forEach(part => {
    const [key, ...rest] = part.split('=')
    if (key && rest.length) parts[key.trim()] = rest.join('=')
  })

  const ts = parts['ts']
  const v1 = parts['v1']
  if (!ts || !v1) return false

  const manifest = `id:${dataId};request-id:${xRequestId};ts:${ts};`
  const expected = crypto
    .createHmac('sha256', secret)
    .update(manifest)
    .digest('hex')

  try {
    return crypto.timingSafeEqual(Buffer.from(v1), Buffer.from(expected))
  } catch {
    return false // length mismatch
  }
}

export async function POST(req: Request) {
  const body = await req.json().catch(() => null)
  if (!body) {
    return NextResponse.json({ error: 'Invalid body' }, { status: 400 })
  }

  const { type, data } = body

  // Only handle payment notifications
  if (type !== 'payment') {
    return NextResponse.json({ received: true, skipped: true })
  }

  const paymentId = String(data?.id ?? '')
  if (!paymentId) {
    return NextResponse.json({ error: 'Missing payment id' }, { status: 400 })
  }

  // D-13: HMAC validation BEFORE any DB write
  const xSignature = req.headers.get('x-signature') ?? ''
  const xRequestId = req.headers.get('x-request-id') ?? ''
  if (!verifySignature(paymentId, xSignature, xRequestId)) {
    return NextResponse.json({ error: 'Invalid signature' }, { status: 401 })
  }

  const accessToken = process.env.MP_ACCESS_TOKEN
  if (!accessToken) {
    console.error('[webhook] MP_ACCESS_TOKEN not configured')
    return NextResponse.json({ received: true })
  }

  // Re-fetch payment from MercadoPago (never trust webhook body for status)
  const { MercadoPagoConfig, Payment } = await import('mercadopago')
  const client = new MercadoPagoConfig({ accessToken })
  const payment = new Payment(client)
  const paymentData = await payment.get({ id: paymentId }).catch(() => null)

  if (!paymentData) {
    return NextResponse.json({ error: 'Payment not found' }, { status: 404 })
  }

  const orderId = paymentData.external_reference
    ? parseInt(paymentData.external_reference, 10)
    : null

  if (!orderId || isNaN(orderId)) {
    return NextResponse.json({ received: true, note: 'No external_reference' })
  }

  // D-12: Idempotency — check if this mercadopagoId was already processed
  const [existing] = await db
    .select({ id: orders.id })
    .from(orders)
    .where(eq(orders.mercadopagoId, paymentId))
    .limit(1)

  if (existing) {
    // Already processed this payment — return 200 (idempotent)
    return NextResponse.json({ received: true, duplicate: true })
  }

  // Map MP status to internal status
  const mpStatus = paymentData.status ?? ''

  if (mpStatus === 'approved') {
    // D-11: Webhook writes to deliveryQueue, NEVER calls RCON directly
    await db.transaction(async (tx) => {
      // Update order: paid + stamp mercadopagoId
      await tx
        .update(orders)
        .set({
          status: 'paid',
          mercadopagoId: paymentId,
          updatedAt: new Date(),
        })
        .where(eq(orders.id, orderId))

      // Fetch order + item + user for RCON command resolution
      const [order] = await tx
        .select({
          id: orders.id,
          shopItemId: orders.shopItemId,
          userId: orders.userId,
        })
        .from(orders)
        .where(eq(orders.id, orderId))
        .limit(1)

      if (!order) return

      const [item] = await tx
        .select({
          type: shopItems.type,
          rconTemplate: shopItems.rconTemplate,
          name: shopItems.name,
          duration: shopItems.duration,
        })
        .from(shopItems)
        .where(eq(shopItems.id, order.shopItemId))
        .limit(1)

      const [user] = await tx
        .select({
          minecraftUuid: users.minecraftUuid,
          minecraftName: users.minecraftName,
        })
        .from(users)
        .where(eq(users.id, order.userId))
        .limit(1)

      if (!item || !user?.minecraftUuid || !user?.minecraftName) return

      const rconCommand = resolveRconCommand(item, user.minecraftName)

      // Insert into delivery queue for cron worker to process
      await tx.insert(deliveryQueue).values({
        orderId: order.id,
        minecraftUuid: user.minecraftUuid,
        rconCommand,
        status: 'pending',
      })
    })
  } else if (mpStatus === 'rejected' || mpStatus === 'cancelled') {
    await db
      .update(orders)
      .set({ status: 'failed', mercadopagoId: paymentId, updatedAt: new Date() })
      .where(eq(orders.id, orderId))
  } else if (mpStatus === 'refunded') {
    await db
      .update(orders)
      .set({ status: 'refunded', mercadopagoId: paymentId, updatedAt: new Date() })
      .where(eq(orders.id, orderId))
  }
  // For 'pending'/'in_process' — do nothing, order stays in pending state

  return NextResponse.json({ received: true, orderId, status: mpStatus })
}

// MercadoPago sends GET requests to verify the webhook URL is reachable
export async function GET() {
  return NextResponse.json({ ok: true })
}
```
