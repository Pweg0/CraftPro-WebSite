```typescript
import { NextResponse } from 'next/server'
import { Rcon } from 'rcon-client'
import { db } from '@/lib/db'
import { deliveryQueue, orders } from '@/lib/db/schema'
import { eq, and, lt, isNull, inArray, sql } from 'drizzle-orm'

export const runtime = 'nodejs'
export const maxDuration = 60 // claim full 60s on Vercel Hobby

const BATCH_LIMIT = 5      // max items per cron run (Vercel timeout safety)
const MAX_AGE_MS = 48 * 60 * 60 * 1000  // 48h — D-09 max retry window

export async function GET(req: Request) {
  // Auth gate: Bearer token from cron-job.org
  const authHeader = req.headers.get('authorization') ?? ''
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  // Step 1: Mark expired items as permanent failure (D-09: 48h max)
  const cutoff = new Date(Date.now() - MAX_AGE_MS)
  await db
    .update(deliveryQueue)
    .set({ status: 'failed', lockedAt: null })
    .where(
      and(
        eq(deliveryQueue.status, 'pending'),
        lt(deliveryQueue.createdAt, cutoff)
      )
    )

  // Also mark the parent orders as failed for expired queue items
  const expiredItems = await db
    .select({ orderId: deliveryQueue.orderId })
    .from(deliveryQueue)
    .where(
      and(
        eq(deliveryQueue.status, 'failed'),
        lt(deliveryQueue.createdAt, cutoff)
      )
    )

  if (expiredItems.length > 0) {
    await db
      .update(orders)
      .set({ status: 'failed', updatedAt: new Date() })
      .where(
        inArray(orders.id, expiredItems.map(e => e.orderId))
      )
  }

  // Step 2: Select and lock pending items (lockedAt = null prevents double-processing)
  const pending = await db
    .select()
    .from(deliveryQueue)
    .where(
      and(
        eq(deliveryQueue.status, 'pending'),
        isNull(deliveryQueue.lockedAt)
      )
    )
    .limit(BATCH_LIMIT)

  if (pending.length === 0) {
    return NextResponse.json({ processed: 0, delivered: 0, failed: 0 })
  }

  // Lock items
  const pendingIds = pending.map(p => p.id)
  await db
    .update(deliveryQueue)
    .set({ lockedAt: new Date() })
    .where(inArray(deliveryQueue.id, pendingIds))

  let delivered = 0
  let failed = 0

  // Step 3: Process each item — connect RCON, send command, update status
  for (const item of pending) {
    let rcon: Rcon | null = null
    try {
      rcon = await Rcon.connect({
        host: process.env.MC_RCON_HOST!,
        port: Number(process.env.MC_RCON_PORT ?? 25575),
        password: process.env.MC_RCON_PASSWORD!,
        timeout: 5000, // 5s per connection — don't hang indefinitely
      })

      await rcon.send(item.rconCommand)
      await rcon.end()

      // Mark delivery as successful
      const now = new Date()
      await db
        .update(deliveryQueue)
        .set({ status: 'delivered', deliveredAt: now, lockedAt: null })
        .where(eq(deliveryQueue.id, item.id))

      // Update parent order to delivered
      await db
        .update(orders)
        .set({ status: 'delivered', deliveredAt: now, updatedAt: now })
        .where(eq(orders.id, item.orderId))

      delivered++
    } catch (error) {
      // RCON failed — unlock item, increment retry, set lastAttemptAt
      try { rcon?.end() } catch { /* ignore close errors */ }

      await db
        .update(deliveryQueue)
        .set({
          retryCount: sql`${deliveryQueue.retryCount} + 1`,
          lastAttemptAt: new Date(),
          lockedAt: null, // unlock for next cron run
        })
        .where(eq(deliveryQueue.id, item.id))

      // Update order status to 'delivering' (in-progress indicator)
      await db
        .update(orders)
        .set({ status: 'delivering', updatedAt: new Date() })
        .where(eq(orders.id, item.orderId))

      failed++
    }
  }

  return NextResponse.json({ processed: pending.length, delivered, failed })
}
```

```typescript
export async function forceRetryDelivery(orderId: number): Promise<void> {
  const session = await auth()
  if (!(await hasPermission(session, 'pedidos'))) throw new Error('Forbidden')

  // Fetch the order and verify it's in a failed state
  const [order] = await db
    .select()
    .from(orders)
    .where(eq(orders.id, orderId))
    .limit(1)

  if (!order) throw new Error('Pedido nao encontrado')
  if (order.status !== 'failed') throw new Error('Pedido nao esta em estado de falha')

  // Fetch shop item for RCON command generation
  const [item] = await db
    .select({
      type: shopItems.type,
      rconTemplate: shopItems.rconTemplate,
      name: shopItems.name,
      duration: shopItems.duration,
    })
    .from(shopItems)
    .where(eq(shopItems.id, order.shopItemId))
    .limit(1)

  if (!item) throw new Error('Item da loja nao encontrado')

  // Fetch user for Minecraft name
  const [user] = await db
    .select({
      minecraftUuid: users.minecraftUuid,
      minecraftName: users.minecraftName,
    })
    .from(users)
    .where(eq(users.id, order.userId))
    .limit(1)

  if (!user?.minecraftUuid || !user?.minecraftName) {
    throw new Error('Jogador sem conta Minecraft vinculada')
  }

  const rconCommand = resolveRconCommand(item, user.minecraftName)

  // Insert new delivery queue entry for the cron worker to pick up
  await db.insert(deliveryQueue).values({
    orderId: order.id,
    minecraftUuid: user.minecraftUuid,
    rconCommand,
    status: 'pending',
  })

  // Update order status to 'delivering' (cron will set to 'delivered' on success)
  await db
    .update(orders)
    .set({ status: 'delivering', updatedAt: new Date() })
    .where(eq(orders.id, orderId))

  await logAudit(session!, 'FORCE_RETRY_ORDER', String(orderId))
  revalidatePath('/admin/pedidos')
}
```
