package com.craftprobrasil.craftpro.api;

import com.craftprobrasil.craftpro.CraftProMod;
import com.craftprobrasil.craftpro.config.CraftProConfig;
import com.craftprobrasil.craftpro.delivery.PendingDelivery;
import com.craftprobrasil.craftpro.delivery.PendingRevocation;
import com.google.gson.Gson;
import com.google.gson.JsonArray;
import com.google.gson.JsonObject;
import com.google.gson.JsonParser;

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;
import java.util.ArrayList;
import java.util.List;

public class SiteApiClient {
    private static final Gson GSON = new Gson();
    private static final HttpClient HTTP_CLIENT = HttpClient.newBuilder()
            .connectTimeout(Duration.ofSeconds(10))
            .build();

    public static class PendingResponse {
        public List<PendingDelivery> orders = new ArrayList<>();
        public List<PendingRevocation> revocations = new ArrayList<>();
    }

    public static PendingResponse fetchPending() throws Exception {
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(CraftProConfig.getApiUrl() + "/api/deliveries/pending"))
                .header("Authorization", "Bearer " + CraftProConfig.getApiKey())
                .GET()
                .timeout(Duration.ofSeconds(15))
                .build();

        HttpResponse<String> response = HTTP_CLIENT.send(request, HttpResponse.BodyHandlers.ofString());
        if (response.statusCode() != 200) {
            throw new RuntimeException("API retornou status " + response.statusCode());
        }

        JsonObject json = JsonParser.parseString(response.body()).getAsJsonObject();
        PendingResponse result = new PendingResponse();

        JsonArray ordersArray = json.getAsJsonArray("orders");
        if (ordersArray != null) {
            for (var elem : ordersArray) {
                result.orders.add(GSON.fromJson(elem, PendingDelivery.class));
            }
        }

        JsonArray revocationsArray = json.getAsJsonArray("revocations");
        if (revocationsArray != null) {
            for (var elem : revocationsArray) {
                result.revocations.add(GSON.fromJson(elem, PendingRevocation.class));
            }
        }

        return result;
    }

    public static void confirmDelivery(int orderId, String expiresAtIso) throws Exception {
        JsonObject body = new JsonObject();
        body.addProperty("orderId", orderId);
        body.addProperty("expiresAt", expiresAtIso);

        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(CraftProConfig.getApiUrl() + "/api/deliveries/confirm"))
                .header("Authorization", "Bearer " + CraftProConfig.getApiKey())
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(GSON.toJson(body)))
                .timeout(Duration.ofSeconds(15))
                .build();

        HttpResponse<String> response = HTTP_CLIENT.send(request, HttpResponse.BodyHandlers.ofString());
        if (response.statusCode() != 200) {
            throw new RuntimeException("Confirm falhou com status " + response.statusCode());
        }
    }

    public static void notifyExpired(int orderId) throws Exception {
        JsonObject body = new JsonObject();
        body.addProperty("orderId", orderId);

        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(CraftProConfig.getApiUrl() + "/api/deliveries/expired"))
                .header("Authorization", "Bearer " + CraftProConfig.getApiKey())
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(GSON.toJson(body)))
                .timeout(Duration.ofSeconds(15))
                .build();

        HttpResponse<String> response = HTTP_CLIENT.send(request, HttpResponse.BodyHandlers.ofString());
        if (response.statusCode() != 200) {
            CraftProMod.LOGGER.warn("CraftPro: notifyExpired falhou com status {}", response.statusCode());
        }
    }
}

package com.craftprobrasil.craftpro.delivery;

import com.craftprobrasil.craftpro.CraftProMod;
import com.craftprobrasil.craftpro.api.SiteApiClient;
import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import com.google.gson.reflect.TypeToken;
import net.minecraft.network.chat.Component;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.level.ServerPlayer;

import net.neoforged.fml.loading.FMLPaths;

import java.io.IOException;
import java.lang.reflect.Type;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.Instant;
import java.time.ZoneOffset;
import java.time.format.DateTimeFormatter;
import java.util.ArrayList;
import java.util.Iterator;
import java.util.List;
import java.util.concurrent.CompletableFuture;

public class BenefitManager {
    private static final Gson GSON = new GsonBuilder().setPrettyPrinting().create();
    private static final Path BENEFITS_FILE = FMLPaths.CONFIGDIR.get().resolve("craftpro/benefits.json");
    private static final List<ActiveBenefit> activeBenefits = new ArrayList<>();

    public static synchronized void loadFromDisk() {
        activeBenefits.clear();
        if (!Files.exists(BENEFITS_FILE)) return;
        try {
            String json = Files.readString(BENEFITS_FILE);
            Type listType = new TypeToken<List<ActiveBenefit>>(){}.getType();
            List<ActiveBenefit> loaded = GSON.fromJson(json, listType);
            if (loaded != null) activeBenefits.addAll(loaded);
            CraftProMod.LOGGER.info("CraftPro: {} beneficios carregados do disco.", activeBenefits.size());
        } catch (Exception e) {
            CraftProMod.LOGGER.error("CraftPro: erro ao carregar benefits.json: {}", e.getMessage());
        }
    }

    public static synchronized void saveToDisk() {
        try {
            Files.createDirectories(BENEFITS_FILE.getParent());
            Files.writeString(BENEFITS_FILE, GSON.toJson(activeBenefits));
        } catch (IOException e) {
            CraftProMod.LOGGER.error("CraftPro: erro ao salvar benefits.json: {}", e.getMessage());
        }
    }

    public static synchronized void addBenefit(ActiveBenefit benefit) {
        activeBenefits.add(benefit);
        saveToDisk();
    }

    public static synchronized void removeBenefit(int orderId) {
        activeBenefits.removeIf(b -> b.orderId == orderId);
        saveToDisk();
    }

    public static synchronized boolean hasBenefit(int orderId) {
        return activeBenefits.stream().anyMatch(b -> b.orderId == orderId);
    }

    public static synchronized int getActiveCount() {
        return activeBenefits.size();
    }

    public static synchronized List<ActiveBenefit> getAll() {
        return new ArrayList<>(activeBenefits);
    }

    /**
     * Check for expired benefits. Called every 60 seconds from TickScheduler.
     * Per D-10: real time expiration (no grace period).
     * Per D-11: executes /ftbranks remove and notifies site.
     * Per D-12: sends chat message to online player.
     */
    public static void checkExpirations(MinecraftServer server) {
        List<ActiveBenefit> expired = new ArrayList<>();
        synchronized (BenefitManager.class) {
            Iterator<ActiveBenefit> it = activeBenefits.iterator();
            while (it.hasNext()) {
                ActiveBenefit b = it.next();
                if (b.isExpired()) {
                    expired.add(b);
                    it.remove();
                }
            }
            if (!expired.isEmpty()) saveToDisk();
        }

        for (ActiveBenefit b : expired) {
            // Execute /ftbranks remove on server thread
            server.execute(() -> {
                String cmd = "ftbranks remove " + b.playerName + " " + b.rankSlug;
                server.getCommands().performPrefixedCommand(
                        server.createCommandSourceStack().withSuppressedOutput(), cmd);
                CraftProMod.LOGGER.info("CraftPro: rank {} removido de {} (order {})", b.rankSlug, b.playerName, b.orderId);

                // D-12: notify player if online
                ServerPlayer player = server.getPlayerList().getPlayerByName(b.playerName);
                if (player != null) {
                    player.sendSystemMessage(Component.literal(
                            "\u00a7c[CraftPro] \u00a7fSeu beneficio \u00a7e" + b.rankSlug + " \u00a7fexpirou. Obrigado por apoiar o servidor!"));
                }
            });

            // Notify site asynchronously
            CompletableFuture.runAsync(() -> {
                try {
                    SiteApiClient.notifyExpired(b.orderId);
                    CraftProMod.LOGGER.info("CraftPro: site notificado sobre expiracao do order {}", b.orderId);
                } catch (Exception e) {
                    CraftProMod.LOGGER.warn("CraftPro: falha ao notificar site sobre expiracao: {}", e.getMessage());
                }
            });
        }
    }
}

    public static void addToQueue(PendingDelivery delivery) {
        // Avoid duplicates
        if (queue.stream().noneMatch(d -> d.orderId == delivery.orderId)) {
            queue.add(delivery);
            CraftProMod.LOGGER.info("CraftPro: order {} adicionado a fila (player {} offline)", delivery.orderId, delivery.minecraftName);
        }
    }

    public static void addRevocation(PendingRevocation revocation) {
        if (revocationQueue.stream().noneMatch(r -> r.orderId == revocation.orderId)) {
            revocationQueue.add(revocation);
            CraftProMod.LOGGER.info("CraftPro: revogacao order {} adicionada a fila", revocation.orderId);
        }
    }

    public static int getPendingCount() {
        return queue.size();
    }

    /**
     * Process deliveries for online players. Called after each poll.
     * Per D-03: only deliver when player is online.
     */
    public static void processQueue(MinecraftServer server) {
        Iterator<PendingDelivery> it = queue.iterator();
        List<PendingDelivery> delivered = new ArrayList<>();

        while (it.hasNext()) {
            PendingDelivery d = it.next();
            ServerPlayer player = server.getPlayerList().getPlayerByName(d.minecraftName);
            if (player != null) {
                deliverToPlayer(server, d);
                delivered.add(d);
            }
        }

        queue.removeAll(delivered);
    }

    /**
     * Process revocations. Remove rank from players who had benefits revoked by admin.
     */
    public static void processRevocations(MinecraftServer server) {
        Iterator<PendingRevocation> it = revocationQueue.iterator();
        List<PendingRevocation> processed = new ArrayList<>();

        while (it.hasNext()) {
            PendingRevocation r = it.next();
            // Execute remove regardless of player online status (FTB Ranks works offline too)
            server.execute(() -> {
                String cmd = "ftbranks remove " + r.minecraftName + " " + r.itemName;
                server.getCommands().performPrefixedCommand(
                        server.createCommandSourceStack().withSuppressedOutput(), cmd);
                CraftProMod.LOGGER.info("CraftPro: rank {} revogado de {} (order {})", r.itemName, r.minecraftName, r.orderId);
            });
            BenefitManager.removeBenefit(r.orderId);
            processed.add(r);
        }

        revocationQueue.removeAll(processed);
    }

    private static void deliverToPlayer(MinecraftServer server, PendingDelivery d) {
        // Determine rank slug from item name/type
        String rankSlug = d.itemName.toLowerCase().replace(" ", "_");

        // Execute /ftbranks add on server thread
        server.execute(() -> {
            String cmd = "ftbranks add " + d.minecraftName + " " + rankSlug;
            server.getCommands().performPrefixedCommand(
                    server.createCommandSourceStack().withSuppressedOutput(), cmd);
            CraftProMod.LOGGER.info("CraftPro: rank {} entregue a {} (order {})", rankSlug, d.minecraftName, d.orderId);

            // Notify player
            ServerPlayer player = server.getPlayerList().getPlayerByName(d.minecraftName);
            if (player != null) {
                player.sendSystemMessage(net.minecraft.network.chat.Component.literal(
                        "\u00a7a[CraftPro] \u00a7fVoce recebeu o beneficio \u00a7e" + d.itemName + "\u00a7f! Obrigado pela compra!"));
            }
        });

        // Calculate expiration
        int durationDays = (d.itemDuration != null && d.itemDuration > 0) ? d.itemDuration : 30;
        Instant now = Instant.now();
        Instant expiresAt = now.plus(durationDays, ChronoUnit.DAYS);
        String expiresAtIso = DateTimeFormatter.ISO_INSTANT.format(expiresAt);

        // Save active benefit locally (per D-08)
        ActiveBenefit benefit = new ActiveBenefit(
                d.minecraftName, d.minecraftUuid, rankSlug, d.orderId,
                now.toEpochMilli(), expiresAt.toEpochMilli());
        BenefitManager.addBenefit(benefit);

        // Confirm to site asynchronously (per MOD-03)
        CompletableFuture.runAsync(() -> {
            try {
                SiteApiClient.confirmDelivery(d.orderId, expiresAtIso);
                CraftProMod.LOGGER.info("CraftPro: entrega do order {} confirmada pro site", d.orderId);
            } catch (Exception e) {
                CraftProMod.LOGGER.warn("CraftPro: falha ao confirmar entrega do order {}: {}", d.orderId, e.getMessage());
            }
        });
    }
}

package com.craftprobrasil.craftpro.scheduler;

import com.craftprobrasil.craftpro.CraftProMod;
import com.craftprobrasil.craftpro.api.SiteApiClient;
import com.craftprobrasil.craftpro.config.CraftProConfig;
import com.craftprobrasil.craftpro.delivery.BenefitManager;
import com.craftprobrasil.craftpro.delivery.DeliveryQueue;
import com.craftprobrasil.craftpro.delivery.PendingDelivery;
import net.minecraft.server.MinecraftServer;
import net.neoforged.bus.api.SubscribeEvent;
import net.neoforged.neoforge.event.tick.ServerTickEvent;

import java.util.concurrent.CompletableFuture;
import java.util.concurrent.atomic.AtomicBoolean;

public class TickScheduler {
    private static int pollTickCounter = 0;
    private static int expiryTickCounter = 0;
    private static final AtomicBoolean polling = new AtomicBoolean(false);

    @SubscribeEvent
    public static void onServerTick(ServerTickEvent.Post event) {
        pollTickCounter++;
        expiryTickCounter++;

        MinecraftServer server = event.getServer();

        // D-02: poll every 30 seconds (600 ticks at 20 TPS)
        if (pollTickCounter >= CraftProConfig.getPollIntervalTicks()) {
            pollTickCounter = 0;

            // Process any queued deliveries for now-online players
            DeliveryQueue.processQueue(server);
            DeliveryQueue.processRevocations(server);

            // Poll API asynchronously (NEVER block tick thread — see Pitfall 1)
            if (!polling.getAndSet(true)) {
                CompletableFuture.runAsync(() -> {
                    try {
                        SiteApiClient.PendingResponse response = SiteApiClient.fetchPending();

                        // Queue new deliveries
                        for (PendingDelivery d : response.orders) {
                            if (!BenefitManager.hasBenefit(d.orderId)) {
                                DeliveryQueue.addToQueue(d);
                            }
                        }

                        // Queue revocations
                        for (var r : response.revocations) {
                            DeliveryQueue.addRevocation(r);
                        }

                        // Process deliveries for online players immediately
                        server.execute(() -> {
                            DeliveryQueue.processQueue(server);
                            DeliveryQueue.processRevocations(server);
                        });

                    } catch (Exception e) {
                        CraftProMod.LOGGER.warn("CraftPro: erro no polling: {}", e.getMessage());
                    } finally {
                        polling.set(false);
                    }
                });
            }
        }

        // D-11: check expirations every 60 seconds (1200 ticks)
        if (expiryTickCounter >= CraftProConfig.getExpiryCheckIntervalTicks()) {
            expiryTickCounter = 0;
            BenefitManager.checkExpirations(server);
        }
    }
}
