1. **Update `middleware.ts`** — add coarse admin check after the existing login redirect. NO DB calls in middleware (Edge runtime). Only env var checks:
```typescript
export default auth((req) => {
  const isLoggedIn = !!req.auth
  const { pathname } = req.nextUrl

  const isProtectedRoute =
    pathname.startsWith("/perfil") ||
    pathname.startsWith("/admin")

  if (!isLoggedIn && isProtectedRoute) {
    const loginUrl = new URL("/login", req.nextUrl.origin)
    loginUrl.searchParams.set("callbackUrl", pathname)
    return NextResponse.redirect(loginUrl)
  }

  // Coarse admin check — env var only, no DB call (Edge runtime)
  if (pathname.startsWith("/admin") && isLoggedIn) {
    const discordId = (req.auth?.user as any)?.discordId
    const superAdminId = process.env.SUPER_ADMIN_DISCORD_ID
    const adminIds = (process.env.ADMIN_DISCORD_IDS ?? '').split(',').map(s => s.trim()).filter(Boolean)
    const isAdminUser = discordId === superAdminId || adminIds.includes(discordId ?? '')
    if (!isAdminUser) {
      return NextResponse.redirect(new URL('/403', req.nextUrl.origin))
    }
  }

  return NextResponse.next()
})
```

1. **Create `src/lib/db/queries/admin.ts`** with two exported functions:

```typescript
import { db } from '@/lib/db'
import { arenaPlayers, arenaMatches, bannedItems, servers, adminAuditLog } from '@/lib/db/schema'
import { count, gte, desc } from 'drizzle-orm'

export async function getDashboardStats() {
  const today = new Date()
  today.setHours(0, 0, 0, 0)
  const todayEpoch = Math.floor(today.getTime() / 1000)

  const [playersResult] = await db.select({ total: count() }).from(arenaPlayers)
  const [matchesTodayResult] = await db.select({ total: count() }).from(arenaMatches)
    .where(gte(arenaMatches.startedAt, todayEpoch))
  const [bannedItemsResult] = await db.select({ total: count() }).from(bannedItems)
  const [serversResult] = await db.select({ total: count() }).from(servers)

  return {
    totalPlayers: playersResult.total,
    matchesToday: matchesTodayResult.total,
    totalBannedItems: bannedItemsResult.total,
    totalServers: serversResult.total,
  }
}

export async function getRecentAuditLogs(limit = 10) {
  return db.select({
    id: adminAuditLog.id,
    discordName: adminAuditLog.discordName,
    action: adminAuditLog.action,
    targetId: adminAuditLog.targetId,
    createdAt: adminAuditLog.createdAt,
  })
    .from(adminAuditLog)
    .orderBy(desc(adminAuditLog.createdAt))
    .limit(limit)
}
```
