```typescript
'use server'
import { auth } from '@/lib/auth'
import { hasPermission, logAudit } from '@/lib/admin/permissions'
import { db } from '@/lib/db'
import { arenaPlayers, arenaMatches } from '@/lib/db/schema'
import { eq } from 'drizzle-orm'
import { revalidatePath } from 'next/cache'

export async function banPlayer(uuid: string) {
  const session = await auth()
  if (!(await hasPermission(session, 'leaderboard'))) throw new Error('Forbidden')
  await db.update(arenaPlayers).set({ banned: 1 }).where(eq(arenaPlayers.uuid, uuid))
  await logAudit(session!, 'BAN_PLAYER', uuid)
  revalidatePath('/admin/leaderboard')
  revalidatePath('/ranking')
}

export async function unbanPlayer(uuid: string) {
  const session = await auth()
  if (!(await hasPermission(session, 'leaderboard'))) throw new Error('Forbidden')
  await db.update(arenaPlayers).set({ banned: 0 }).where(eq(arenaPlayers.uuid, uuid))
  await logAudit(session!, 'UNBAN_PLAYER', uuid)
  revalidatePath('/admin/leaderboard')
  revalidatePath('/ranking')
}

export async function deleteMatch(id: number) {
  const session = await auth()
  if (!(await hasPermission(session, 'leaderboard'))) throw new Error('Forbidden')
  await db.delete(arenaMatches).where(eq(arenaMatches.id, id))
  await logAudit(session!, 'DELETE_MATCH', String(id))
  revalidatePath('/admin/leaderboard')
  revalidatePath('/ranking')
}
```

**1. Create `src/lib/admin/actions/permissions.ts`:**

```typescript
'use server'
import { auth } from '@/lib/auth'
import { isSuperAdmin, logAudit } from '@/lib/admin/permissions'
import { db } from '@/lib/db'
import { adminPermissions } from '@/lib/db/schema'
import { eq, and } from 'drizzle-orm'
import { revalidatePath } from 'next/cache'

export async function grantPermission(discordId: string, section: string) {
  const session = await auth()
  if (!isSuperAdmin(session)) throw new Error('Forbidden — super-admin only')

  // Check if permission already exists
  const [existing] = await db.select().from(adminPermissions)
    .where(and(eq(adminPermissions.discordId, discordId), eq(adminPermissions.section, section)))
    .limit(1)
  if (existing) return // already granted

  await db.insert(adminPermissions).values({
    discordId,
    section,
    grantedBy: session!.user!.discordId!,
  })
  await logAudit(session!, 'GRANT_PERMISSION', discordId, { section })
  revalidatePath('/admin/permissoes')
}

export async function revokePermission(discordId: string, section: string) {
  const session = await auth()
  if (!isSuperAdmin(session)) throw new Error('Forbidden — super-admin only')

  await db.delete(adminPermissions)
    .where(and(eq(adminPermissions.discordId, discordId), eq(adminPermissions.section, section)))
  await logAudit(session!, 'REVOKE_PERMISSION', discordId, { section })
  revalidatePath('/admin/permissoes')
}
```

```typescript
      />
    </div>
  )
}
```
