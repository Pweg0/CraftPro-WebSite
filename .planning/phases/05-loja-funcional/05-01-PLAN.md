```typescript
export const orders = pgTable("orders", {
  id: integer("id").primaryKey().generatedAlwaysAsIdentity(),
  userId: text("user_id").notNull().references(() => users.id, { onDelete: "cascade" }),
  shopItemId: integer("shop_item_id").notNull().references(() => shopItems.id),
  status: text("status").notNull().default("pending"), // pending | paid | delivered | failed | refunded
  mercadopagoId: text("mercadopago_id"),               // MP payment ID (filled after checkout)
  amountCents: integer("amount_cents").notNull(),       // snapshot of price at time of order
  createdAt: timestamp("created_at", { mode: "date" }).defaultNow(),
  updatedAt: timestamp("updated_at", { mode: "date" }).defaultNow(),
}, (table) => ({
  userIdx: index("idx_orders_user").on(table.userId),
  statusIdx: index("idx_orders_status").on(table.status),
}))
```

3. Create src/lib/db/queries/shop.ts:
```typescript
import { db } from '@/lib/db'
import { shopItems, orders } from '@/lib/db/schema'
import { eq, and, asc, desc } from 'drizzle-orm'

export async function getShopItems() {
  return db
    .select()
    .from(shopItems)
    .where(eq(shopItems.available, 1))
    .orderBy(asc(shopItems.category), asc(shopItems.name))
}

export async function getShopItemById(id: number) {
  const [item] = await db
    .select()
    .from(shopItems)
    .where(and(eq(shopItems.id, id), eq(shopItems.available, 1)))
    .limit(1)
  return item ?? null
}

export async function getOrdersByUser(userId: string) {
  return db
    .select({
      id: orders.id,
      shopItemId: orders.shopItemId,
      status: orders.status,
      amountCents: orders.amountCents,
      createdAt: orders.createdAt,
    })
    .from(orders)
    .where(eq(orders.userId, userId))
    .orderBy(desc(orders.createdAt))
}

export async function getAllShopItemsAdmin() {
  return db
    .select()
    .from(shopItems)
    .orderBy(asc(shopItems.category), asc(shopItems.name))
}
```

```tsx
// src/components/shop/ShopItemCard.tsx
import { Badge } from '@/components/ui/badge'
import { ShoppingCart } from 'lucide-react'

interface ShopItemCardProps {
  id: number
  name: string
  description: string
  category: string
  price: number       // in cents
  imageUrl?: string | null
}

function formatBRL(cents: number): string {
  return new Intl.NumberFormat('pt-BR', {
    style: 'currency',
    currency: 'BRL',
  }).format(cents / 100)
}

const CATEGORY_COLORS: Record<string, { text: string; border: string; bg: string }> = {
  cosmeticos: { text: 'text-brand-yellow', border: 'border-brand-yellow/30', bg: 'bg-brand-yellow/10' },
  'ranks-vip': { text: 'text-brand-accent', border: 'border-brand-accent/30', bg: 'bg-brand-accent/10' },
  'itens-servidor': { text: 'text-brand-blue', border: 'border-brand-blue/30', bg: 'bg-brand-blue/10' },
}

export function ShopItemCard({ name, description, category, price, imageUrl }: ShopItemCardProps) {
  const colors = CATEGORY_COLORS[category] ?? { text: 'text-white/60', border: 'border-white/20', bg: 'bg-white/5' }

  return (
    <div className="card-glow bg-brand-card/60 backdrop-blur-sm rounded-2xl border border-white/[0.06] overflow-hidden flex flex-col transition-all duration-300 hover:-translate-y-1 hover:border-white/[0.12]">
      {imageUrl && (
        <div className="w-full h-36 bg-brand-card/80 overflow-hidden">
          {/* eslint-disable-next-line @next/next/no-img-element */}
          <img src={imageUrl} alt={name} className="w-full h-full object-cover" />
        </div>
      )}
      <div className="p-5 flex flex-col flex-1">
        <div className="flex items-start justify-between gap-2 mb-2">
          <h3 className="text-sm font-semibold text-white leading-snug">{name}</h3>
          <Badge
            variant="outline"
            className={`text-xs shrink-0 ${colors.text} ${colors.border} ${colors.bg}`}
          >
            {category}
          </Badge>
        </div>
        <p className="text-xs text-white/45 leading-relaxed mb-4 flex-1">{description}</p>
        <div className="flex items-center justify-between">
          <span className="text-lg font-bold text-white">{formatBRL(price)}</span>
          <button
            disabled
            className="inline-flex items-center gap-1.5 text-xs font-medium px-3 py-1.5 rounded-lg bg-brand-accent/20 text-brand-accent border border-brand-accent/30 opacity-60 cursor-not-allowed"
            title="Pagamento em breve"
          >
            <ShoppingCart className="w-3.5 h-3.5" />
            Comprar
          </button>
        </div>
      </div>
    </div>
  )
}
```
