4. Create src/lib/db/queries/blog.ts — follow same pattern as shop.ts:
```typescript
import { db } from '@/lib/db'
import { blogPosts } from '@/lib/db/schema'
import { desc, eq, isNotNull, and, asc } from 'drizzle-orm'

export async function getPublishedPosts() {
  return db.select().from(blogPosts)
    .where(isNotNull(blogPosts.publishedAt))
    .orderBy(desc(blogPosts.pinned), desc(blogPosts.publishedAt))
}

export async function getPostBySlug(slug: string) {
  const [post] = await db.select().from(blogPosts)
    .where(and(eq(blogPosts.slug, slug), isNotNull(blogPosts.publishedAt)))
    .limit(1)
  return post ?? null
}

export async function getAllPostsAdmin() {
  return db.select().from(blogPosts).orderBy(desc(blogPosts.createdAt))
}

export async function getPostById(id: number) {
  const [post] = await db.select().from(blogPosts)
    .where(eq(blogPosts.id, id)).limit(1)
  return post ?? null
}
```

5. Create src/lib/db/queries/events.ts:
```typescript
import { db } from '@/lib/db'
import { events } from '@/lib/db/schema'
import { desc, eq, gte, lte, and, asc, sql } from 'drizzle-orm'

export async function getUpcomingEvents() {
  return db.select().from(events)
    .where(gte(events.startDate, new Date()))
    .orderBy(asc(events.startDate))
}

export async function getEventsByMonth(year: number, month: number) {
  const startOfMonth = new Date(year, month - 1, 1)
  const endOfMonth = new Date(year, month, 0, 23, 59, 59)
  return db.select().from(events)
    .where(and(
      gte(events.startDate, startOfMonth),
      lte(events.startDate, endOfMonth)
    ))
    .orderBy(asc(events.startDate))
}

export async function getAllEventsAdmin() {
  return db.select().from(events).orderBy(desc(events.startDate))
}

export async function getEventById(id: number) {
  const [event] = await db.select().from(events)
    .where(eq(events.id, id)).limit(1)
  return event ?? null
}
```
