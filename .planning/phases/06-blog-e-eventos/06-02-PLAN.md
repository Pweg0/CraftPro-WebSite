```typescript
'use server'
import { auth } from '@/lib/auth'
import { hasPermission, logAudit } from '@/lib/admin/permissions'
import { db } from '@/lib/db'
import { blogPosts } from '@/lib/db/schema'
import { eq } from 'drizzle-orm'
import { revalidatePath } from 'next/cache'

interface BlogPostData {
  title: string
  slug: string
  body: string          // markdown content from editor
  summary?: string | null
  thumbnail?: string | null
  tags: string[]
  pinned: boolean       // per D-03/D-10
  publishedAt: Date | null  // null = draft per D-02
}

export async function createBlogPost(data: BlogPostData) {
  const session = await auth()
  if (!(await hasPermission(session, 'blog'))) throw new Error('Forbidden')
  const [newPost] = await db.insert(blogPosts).values({
    title: data.title,
    slug: data.slug,
    body: data.body,
    summary: data.summary ?? null,
    thumbnail: data.thumbnail ?? null,
    tags: data.tags,
    pinned: data.pinned,
    publishedAt: data.publishedAt,
    authorDiscordId: session!.user!.discordId!,
    authorName: session!.user?.name ?? null,
  }).returning()
  await logAudit(session!, 'CREATE_BLOG_POST', String(newPost.id), { title: data.title, slug: data.slug })
  revalidatePath('/admin/blog')
  revalidatePath('/blog')
  return newPost
}

export async function updateBlogPost(id: number, data: BlogPostData) {
  const session = await auth()
  if (!(await hasPermission(session, 'blog'))) throw new Error('Forbidden')
  const [updated] = await db.update(blogPosts).set({
    title: data.title,
    slug: data.slug,
    body: data.body,
    summary: data.summary ?? null,
    thumbnail: data.thumbnail ?? null,
    tags: data.tags,
    pinned: data.pinned,
    publishedAt: data.publishedAt,
    updatedAt: new Date(),
  }).where(eq(blogPosts.id, id)).returning()
  await logAudit(session!, 'UPDATE_BLOG_POST', String(id), { title: data.title })
  revalidatePath('/admin/blog')
  revalidatePath('/blog')
  revalidatePath(`/blog/${data.slug}`)
  return updated
}

export async function deleteBlogPost(id: number) {
  const session = await auth()
  if (!(await hasPermission(session, 'blog'))) throw new Error('Forbidden')
  await db.delete(blogPosts).where(eq(blogPosts.id, id))
  await logAudit(session!, 'DELETE_BLOG_POST', String(id))
  revalidatePath('/admin/blog')
  revalidatePath('/blog')
}
```
