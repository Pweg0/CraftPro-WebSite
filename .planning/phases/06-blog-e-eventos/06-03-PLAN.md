1. Create src/lib/admin/actions/events.ts following the EXACT pattern from shop-items.ts (per D-13, D-15):

```typescript
'use server'
import { auth } from '@/lib/auth'
import { hasPermission, logAudit } from '@/lib/admin/permissions'
import { db } from '@/lib/db'
import { events } from '@/lib/db/schema'
import { eq } from 'drizzle-orm'
import { revalidatePath } from 'next/cache'

interface EventData {
  title: string
  description: string
  type: string         // 'torneio' | 'update' | 'manutencao' | 'especial' per D-06
  startDate: Date
  endDate: Date | null  // nullable per D-08
  eventUrl: string | null  // nullable per D-08
}

export async function createEvent(data: EventData) {
  const session = await auth()
  if (!(await hasPermission(session, 'eventos'))) throw new Error('Forbidden')
  const [newEvent] = await db.insert(events).values({
    title: data.title,
    description: data.description,
    type: data.type,
    startDate: data.startDate,
    endDate: data.endDate,
    eventUrl: data.eventUrl,
  }).returning()
  await logAudit(session!, 'CREATE_EVENT', String(newEvent.id), { title: data.title, type: data.type })
  revalidatePath('/admin/eventos')
  revalidatePath('/eventos')
  return newEvent
}

export async function updateEvent(id: number, data: EventData) {
  const session = await auth()
  if (!(await hasPermission(session, 'eventos'))) throw new Error('Forbidden')
  const [updated] = await db.update(events).set({
    title: data.title,
    description: data.description,
    type: data.type,
    startDate: data.startDate,
    endDate: data.endDate,
    eventUrl: data.eventUrl,
    updatedAt: new Date(),
  }).where(eq(events.id, id)).returning()
  await logAudit(session!, 'UPDATE_EVENT', String(id), { title: data.title, type: data.type })
  revalidatePath('/admin/eventos')
  revalidatePath('/eventos')
  return updated
}

export async function deleteEvent(id: number) {
  const session = await auth()
  if (!(await hasPermission(session, 'eventos'))) throw new Error('Forbidden')
  await db.delete(events).where(eq(events.id, id))
  await logAudit(session!, 'DELETE_EVENT', String(id))
  revalidatePath('/admin/eventos')
  revalidatePath('/eventos')
}
```
